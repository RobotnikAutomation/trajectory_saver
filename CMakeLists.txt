cmake_minimum_required(VERSION 3.8)
project(trajectory_saver)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(manipulation_msgs REQUIRED)
find_package(warehouse_ros REQUIRED)
add_executable(trajectory_saver_node src/trajectory_saver_node.cpp src/trajectory_saver.cpp)
target_compile_features(trajectory_saver_node PUBLIC cxx_std_17)
target_include_directories(trajectory_saver_node PRIVATE include)
ament_target_dependencies(trajectory_saver_node rclcpp std_msgs std_srvs manipulation_msgs warehouse_ros)
install(TARGETS trajectory_saver_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION include/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_warehouse_store test/test_warehouse_store.cpp)
  target_compile_features(test_warehouse_store PUBLIC cxx_std_17)
  target_include_directories(test_warehouse_store PRIVATE include)
  ament_target_dependencies(test_warehouse_store rclcpp std_msgs manipulation_msgs warehouse_ros)
endif()
ament_package()
